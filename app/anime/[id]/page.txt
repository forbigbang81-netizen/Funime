import Link from "next/link";

const animeDatabase = {
  "solo-leveling": {
    title: "Solo Leveling",
    image: "/anime/solo-leveling.jpg",
    rating: "9.1",
    episodes: 24,
    genres: ["Action", "Fantasy"],
    synopsis:
      "After surviving a deadly dungeon, Sung Jinwoo gains the ability to level up beyond human limits.",
  },

  "jujutsu-kaisen": {
    title: "Jujutsu Kaisen",
    image: "/anime/jjk.jpg",
    rating: "8.9",
    episodes: 47,
    genres: ["Action", "Supernatural"],
    synopsis:
      "Yuji Itadori becomes involved in the world of curses after consuming a powerful cursed object.",
  },
};

type PageProps = {
  params: Promise<{
    id: string;
  }>;
};

export default async function AnimeDetailPage({
  params,
}: PageProps) {
  const { id } = await params;

  const anime =
    animeDatabase[id as keyof typeof animeDatabase];

  if (!anime) {
    return (
      <main className="min-h-screen bg-[#080510] text-white p-10">
        <h1 className="text-4xl font-bold">
          Anime Not Found
        </h1>
      </main>
    );
  }

  return (
    <main className="bg-[#080510] min-h-screen text-white">

      <section className="relative h-[500px]">
        <img
          src={anime.image}
          alt={anime.title}
          className="absolute inset-0 h-full w-full object-cover"
        />

        <div className="absolute inset-0 bg-gradient-to-t from-[#080510] via-[#080510aa] to-transparent" />

        <div className="absolute bottom-12 left-12">
          <h1 className="text-6xl font-bold">
            {anime.title}
          </h1>

          <div className="mt-4 flex gap-4">
            <span>⭐ {anime.rating}</span>
            <span>{anime.episodes} Episodes</span>
          </div>

          <div className="mt-4 flex gap-2">
            {anime.genres.map((genre) => (
              <span
                key={genre}
                className="rounded-full bg-purple-700 px-3 py-1 text-sm"
              >
                {genre}
              </span>
            ))}
          </div>

          <div className="mt-8 flex gap-4">
            <button className="rounded-xl bg-purple-600 px-8 py-3 font-semibold">
              Watch Now
            </button>

            <button className="rounded-xl border border-white px-8 py-3">
              Add to Watchlist
            </button>
          </div>
        </div>
      </section>

      <section className="mx-auto max-w-6xl px-6 py-10">

        <h2 className="text-3xl font-bold">
          Synopsis
        </h2>

        <p className="mt-4 text-gray-300 leading-7">
          {anime.synopsis}
        </p>

        <div className="mt-12">

          <h2 className="text-3xl font-bold">
            Episodes
          </h2>

          <div className="mt-6 space-y-4">

            {Array.from({
              length: anime.episodes,
            }).map((_, index) => (
              <Link
                key={index}
                href={`/watch/${id}/${index + 1}`}
              >
                <div className="rounded-xl bg-[#141021] p-5 transition hover:bg-[#1e1830]">
                  <div className="flex items-center justify-between">
                    <span>
                      Episode {index + 1}
                    </span>

                    <span className="text-purple-400">
                      ▶ Watch
                    </span>
                  </div>
                </div>
              </Link>
            ))}

          </div>
        </div>

        <div className="mt-16">

          <h2 className="text-3xl font-bold">
            Community Discussion
          </h2>

          <div className="mt-6 rounded-xl bg-[#141021] p-5">

            <div className="mb-4 rounded-lg bg-[#1d1830] p-4">
              <div className="mb-2 font-semibold">
                AnimeFan99
              </div>

              <div className="blur-sm hover:blur-none transition">
                Major plot discussion hidden to prevent spoilers.
              </div>
            </div>

            <div className="rounded-lg bg-[#1d1830] p-4">
              <div className="mb-2 font-semibold">
                FunimeUser
              </div>

              <div>
                Incredible animation this season.
              </div>
            </div>

          </div>

        </div>

      </section>

    </main>
  );
}
